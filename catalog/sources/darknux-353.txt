      <div class="status" id="status">Downloading project...</div>

      <div class="track">
        <div class="bar" id="bar"></div>
      </div>

      <div class="info">
        <span id="size">0 MB</span>
        <span id="percent">0%</span>
      </div>
    </div>

  <script>
    const ZIP =
      "https://quizizz.com/_media/uploadedFiles/c0886f20-4198-4637-9a26-332f2c329d81-v2";

    const $ = id => document.getElementById(id);

    const mb = n =>
      (n / 1024 / 1024).toFixed(2) + " MB";

    async function load() {
      const res = await fetch(ZIP);

      if (!res.ok)
        throw new Error("Download failed: " + res.status);

      const total =
        Number(res.headers.get("content-length")) || 0;

      const reader = res.body.getReader();
      const chunks = [];
      let loaded = 0;

      while (true) {
        const { done, value } = await reader.read();
        if (done) break;

        chunks.push(value);
        loaded += value.length;

        if (total) {
          const p = Math.round(loaded / total * 100);
          $("bar").style.width = p + "%";
          $("percent").textContent = p + "%";
          $("size").textContent =
            mb(loaded) + " / " + mb(total);
        } else {
          $("size").textContent = mb(loaded);
        }
      }

      $("title").textContent = "Extracting...";
      $("status").textContent = "Reading project...";

      const data = new Uint8Array(loaded);
      let offset = 0;

      for (const chunk of chunks) {
        data.set(chunk, offset);
        offset += chunk.length;
      }

      const zip = await JSZip.loadAsync(data);

      const htmlFile = Object.values(zip.files)
        .find(f => !f.dir && /\.html?$/i.test(f.name));

      if (!htmlFile)
        throw new Error("No HTML file found.");

      $("title").textContent = "Starting game...";
      $("status").textContent = htmlFile.name;

      const html = await htmlFile.async("blob");

      const url = URL.createObjectURL(
        new Blob([html], { type: "text/html" })
      );

      location.replace(url);
    }

    load().catch(err => {
      console.error(err);

      $("title").textContent = "Failed to load";
      $("status").textContent = err.message;
      $("bar").style.background = "#e44";
    });
  </script>
