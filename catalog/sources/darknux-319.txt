<script>
const BASE_URL="https://quizizz.com/_media/uploads/";

const PACK_URL=BASE_URL+"0ee6b4c9-c614-4455-8434-eb9d2c689cf1-v2";
const WASM_URL=BASE_URL+"f5880ba4-0d65-49f7-8db3-d6edd989e438-v2";

const GODOT_CONFIG={
	args:[],
	canvasResizePolicy:2,
	executable:"index",
	experimentalVK:false,
	focusCanvas:true,
	gdnativeLibs:[],
	mainPack:PACK_URL,
	fileSizes:{
		[PACK_URL]:42459792,
		[WASM_URL]:13793958
	}
};

const status=document.getElementById("status");
const spinner=document.getElementById("spinner");
const progress=document.getElementById("progress");
const progressBar=progress.firstElementChild;
const error=document.getElementById("error");

progress.style.display="none";
error.style.display="none";

const originalFetch=window.fetch;

window.fetch=function(resource,init){
	const url=typeof resource==="string"
		?resource
		:resource.url;

	if(url?.endsWith("index.wasm"))
		resource=WASM_URL;

	return originalFetch.call(this,resource,init);
};

function showStatus(mode){
	spinner.style.display=mode==="spinner"?"block":"none";
	progress.style.display=mode==="progress"?"block":"none";
	error.style.display=mode==="error"?"block":"none";

	if(mode==="hidden"){
		status.style.visibility="hidden";
		status.style.display="none";
	}else{
		status.style.visibility="visible";
		status.style.display="flex";
	}
}

function showError(errorMessage){
	error.textContent=errorMessage.message||errorMessage;
	console.error(errorMessage);
	showStatus("error");
}

if(!Engine.isWebGLAvailable()){
	showError("WebGL not available");
}else{
	showStatus("spinner");

	const engine=new Engine(GODOT_CONFIG);

	engine.startGame({
		onProgress(current,total){
			if(total>0){
				progressBar.style.width=current/total*100+"%";
				showStatus("progress");

				if(current===total){
					setTimeout(()=>showStatus("spinner"),500);
				}
			}else{
				showStatus("spinner");
			}
		}
	}).then(
		()=>{
			status.style.visibility="hidden";
			status.style.display="none";
		},
		showError
	);
}
</script>
