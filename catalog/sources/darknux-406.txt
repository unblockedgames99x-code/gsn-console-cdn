        "https://cdn.jsdelivr.net/gh/shayderrr/eaglercraft@main/tuff/clTuff_Client_Offline_WASM.html.part1",

      for (const u of urls) {
        const res = await fetch(u);
        fullHtml += await res.text();
      }

      document.open();
      document.write(fullHtml);
      document.close();
    }
