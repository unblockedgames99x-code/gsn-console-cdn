    const canvas = document.querySelector("#unity-canvas");
    const loadingBar = document.querySelector("#unity-loading-bar");
    const progressBar = document.querySelector("#unity-progress-bar-full");
    const warningBanner = document.querySelector("#unity-warning");

    function unityShowBanner(message, type) {
      const div = document.createElement("div");
      div.innerHTML = message;

      if (type === "error") {
        div.style.cssText = "background:red;padding:10px;";
      } else if (type === "warning") {
        div.style.cssText = "background:yellow;padding:10px;";
      }

      warningBanner.appendChild(div);
      warningBanner.style.display = "block";

      if (type !== "error") {
        setTimeout(() => {
          div.remove();

          if (!warningBanner.children.length) {
            warningBanner.style.display = "none";
          }
        }, 5000);
      }
    }

    const config = {
      dataUrl:
        "https://quizizz.com/_media/baldi/1fc8ad1a-50d8-4bfa-860c-9cf2d49bfca3-v2",
      frameworkUrl:
        "https://quizizz.com/_media/uploads/8d348646-99d8-4bdf-a1db-11ae03797dd1-v2",
      codeUrl:
        "https://quizizz.com/_media/uploads/7a553bbe-9282-4aff-ac12-704b5a8671cf-v2",
      streamingAssetsUrl: "StreamingAssets",
      companyName: "Naberson265 Games",
      productName: "Baldi's Basics Squared Demo",
      productVersion: "FD1.0d",
      showBanner: unityShowBanner
    };

    canvas.style.background =

    const script = document.createElement("script");
    script.src =
      "https://quizizz.com/_media/uploads/97b12f1c-9909-40c3-8e7e-e6cb3ff21988-v2";

    script.onload = () => {
      createUnityInstance(canvas, config, progress => {
        progressBar.style.transform = `scaleX(${progress})`;
      })
        .then(() => {
          loadingBar.style.display = "none";
        })
        .catch(error => {
          alert(error);
        });
    };

    document.body.appendChild(script);
