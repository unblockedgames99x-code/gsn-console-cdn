  <script>
    const canvas = document.querySelector("#unity-canvas");
    const loadingBar = document.querySelector("#unity-loading-bar");
    const progressBar = document.querySelector("#unity-progress-bar-full");
    const warningBanner = document.querySelector("#unity-warning");

    function unityShowBanner(message, type) {
      const div = document.createElement("div");
      div.innerHTML = message;

      if (type === "error") {
        div.style = "background:red;padding:10px;";
      } else if (type === "warning") {
        div.style = "background:yellow;padding:10px;";

        setTimeout(() => {
          div.remove();

          if (!warningBanner.children.length) {
            warningBanner.style.display = "none";
          }
        }, 5000);
      }

      warningBanner.appendChild(div);
      warningBanner.style.display = "block";
    }

    const loaderUrl =
      "https://quizizz.com/_media/uploads/895a2e42-afde-4f6a-b93c-f51dc8f990d2-v2";

    const config = {
      arguments: [],
      dataUrl:
        "https://quizizz.com/_media/uploads/5cc72d4b-f2b8-440c-8c48-9aea1e11e193-v2",
      frameworkUrl:
        "https://quizizz.com/_media/uploads/6ec59cea-b81e-489d-a9dd-b6e5a3dad831-v2",
      codeUrl:
        "https://quizizz.com/_media/uploads/894d3770-a98f-4361-a0d0-4a468cd34e4c-v2",
      streamingAssetsUrl: "StreamingAssets",
      companyName: "None",
      productName: "Baldi's Basics Squared",
      productVersion: "0.0.7f",
      showBanner: unityShowBanner
    };

    if (/iPhone|iPad|iPod|Android/i.test(navigator.userAgent)) {
      const meta = document.createElement("meta");

      meta.name = "viewport";
      meta.content =
        "width=device-width,height=device-height,initial-scale=1.0,user-scalable=no";

      document.head.appendChild(meta);

      document.querySelector("#unity-container").className = "unity-mobile";
      canvas.className = "unity-mobile";
    }

    loadingBar.style.display = "block";

    const script = document.createElement("script");
    script.src = loaderUrl;

    script.onload = () => {
      createUnityInstance(canvas, config, progress => {
        progressBar.style.width = `${100 * progress}%`;
      })
        .then(() => {
          loadingBar.style.display = "none";
        })
        .catch(error => {
          alert(error);
        });
    };

    document.body.appendChild(script);
  </script>
