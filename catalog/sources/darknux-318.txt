<script>
const BASE_URL="https://quizizz.com/_media/uploads/";

const PACK_URL=BASE_URL+"dfb79abe-fda1-405c-b290-724531b82c17-v2";
const WASM_URL=BASE_URL+"142cf22f-7601-49e5-aeea-700476fcee36-v2";

const GODOT_CONFIG={
	args:[],
	canvasResizePolicy:2,
	executable:"index",
	experimentalVK:false,
	focusCanvas:true,
	gdnativeLibs:[],
	mainPack:PACK_URL,
	fileSizes:{
		[PACK_URL]:42459792,
		[WASM_URL]:13793958
	}
};

const originalFetch=window.fetch;

window.fetch=function(resource,init){
	const url=typeof resource==="string"
		?resource
		:resource.url;

	if(url?.endsWith("index.wasm"))
		resource=WASM_URL;

	return originalFetch.call(this,resource,init);
};

const status=document.getElementById("status");
const spinner=document.getElementById("spinner");
const progress=document.getElementById("progress");
const bar=progress.firstElementChild;
const error=document.getElementById("error");

progress.style.display="none";
error.style.display="none";

function show(mode){
	spinner.style.display=mode==="spinner"?"block":"none";
	progress.style.display=mode==="progress"?"block":"none";
	error.style.display=mode==="error"?"block":"none";

	if(mode==="hidden"){
		status.style.visibility="hidden";
	}else{
		status.style.visibility="visible";
	}
}

function fail(err){
	error.textContent=err.message||err;
	console.error(err);
	show("error");
}

if(!Engine.isWebGLAvailable()){
	fail("WebGL not available");
}else{
	show("spinner");

	const engine=new Engine(GODOT_CONFIG);

	engine.startGame({
		onProgress(current,total){
			if(total>0){
				bar.style.width=current/total*100+"%";
				show("progress");

				if(current===total){
					setTimeout(()=>show("spinner"),500);
				}
			}else{
				show("spinner");
			}
		}
	}).then(
		()=>{
			status.style.visibility="hidden";
			status.style.display="none";
		},
		fail
	);
}
</script>
