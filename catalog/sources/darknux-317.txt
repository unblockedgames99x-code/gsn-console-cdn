  <script>
    const canvas = document.querySelector("#unity-canvas");
    const loadingBar = document.querySelector("#unity-loading-bar");
    const progressBar = document.querySelector("#unity-progress-bar-full");
    const warningBanner = document.querySelector("#unity-warning");

    function unityShowBanner(message, type) {
      const div = document.createElement("div");
      div.innerHTML = message;

      if (type === "error") {
        div.style.cssText = "background:red;padding:10px;";
      } else if (type === "warning") {
        div.style.cssText = "background:yellow;padding:10px;";
      }

      warningBanner.appendChild(div);
      warningBanner.style.display = "block";

      if (type !== "error") {
        setTimeout(() => {
          div.remove();

          if (!warningBanner.children.length) {
            warningBanner.style.display = "none";
          }
        }, 5000);
      }
    }

    const config = {
      dataUrl:
        "https://quizizz.com/_media/uploads/51e44301-d464-419b-8db3-2b006d83f658-v2",
      frameworkUrl:
        "https://quizizz.com/_media/uploads/ab05cb05-4153-4ec7-ac16-2ca8039b9050-v2",
      codeUrl:
        "https://quizizz.com/_media/uploads/ad7c873c-4d08-48df-a41f-f7df509a8525-v2",
      streamingAssetsUrl: "StreamingAssets",
      companyName: "ItzMaxwell",
      productName: "Baldis Basics In Wonky Physics",
      productVersion: "eh, who cares about the version -_-",
      showBanner: unityShowBanner
    };

    const script = document.createElement("script");

    script.src =
      "https://quizizz.com/_media/uploads/2bf7f12d-220f-4136-b5e1-ad69ba6392e2-v2";

    loadingBar.style.display = "block";

    script.onload = () => {
      createUnityInstance(
        canvas,
        config,
        progress => {
          progressBar.style.width = `${100 * progress}%`;
        }
      )
        .then(() => {
          loadingBar.style.display = "none";
        })
        .catch(message => {
          unityShowBanner(message, "error");
        });
    };

    script.onerror = () => {
      loadingBar.style.display = "none";
      unityShowBanner("Failed to load Unity loader script.", "error");
    };

    document.body.appendChild(script);
  </script>
